```sql
-- REPLICA (one row per resolved (event, key) pair, with status as attribute() would classify it)
with p as (select '2026-10-04T18:15:51.251Z'::timestamptz t, '2026-10-04 18:15:55.824614+00'::timestamptz hi),
pin as (select jsonb_array_elements_text(generation_metadata->'node_set'->'keys') key
        from weave_profile_snapshots where id='ce322885-cf00-4cda-8a8b-ad193886cf14'),
emb as (select board_id||':'||node_id key, archived_at from weave_embeddings),
qs as (select distinct e.session_id from weave_events e join voice_sessions vs on vs.id=e.voice_session_id where vs.session_kind='qa'),
ev as (
  select e.id::text ev_id, e.event_type, e.timestamp at_, e.session_id bsess, e.duration_ms, null::int turns,
    case when e.event_type='item_added' then 'recency' else 'breadth' end cls,
    case when e.event_type in ('lightbox_closed','item_added') then
           case when e.target_id like 'node:%' then array[substr(e.target_id,6)] else array[]::text[] end
         when array_length(string_to_array(e.target_id,':'),1)=4 and split_part(e.target_id,':',1)='connection'
           then array[split_part(e.target_id,':',2)||':'||split_part(e.target_id,':',3), split_part(e.target_id,':',2)||':'||split_part(e.target_id,':',4)]
         else array[]::text[] end keys,
    case when e.event_type='item_added' then 0.2
         else case when coalesce(e.duration_ms,0)<=0 then 0 else 1.5*least((ln(e.duration_ms/1000.0+1)/ln(2))/(ln(46)/ln(2)),1) end end w_rule,
    14.0 h, e.target_id, (e.session_id in (select session_id from qs)) qa
  from weave_events e, p
  where e.event_type in ('lightbox_closed','connection_description_closed','item_added')
    and e.timestamp >= p.t - interval '70 days' and e.timestamp <= p.hi
  union all
  select vs.id::text, 'voice_session', vs.ended_at, null, null,
    (select count(*) from voice_utterances u where u.session_id=vs.id and u.speaker='user')::int, 'depth',
    array[ed.board_id||':'||coalesce(fn.data->>'_clientNodeId',fn.id::text), ed.board_id||':'||coalesce(tn.data->>'_clientNodeId',tn.id::text)],
    (1.5/(ln(5)/ln(2)))*ln((select count(*) from voice_utterances u where u.session_id=vs.id and u.speaker='user')+1)/ln(2),
    42.0, 'connection:'||ed.board_id||':'||coalesce(fn.data->>'_clientNodeId',fn.id::text)||':'||coalesce(tn.data->>'_clientNodeId',tn.id::text), false
  from voice_sessions vs join edges ed on ed.id=vs.anchor_edge_id join nodes fn on fn.id=ed.source_node_id join nodes tn on tn.id=ed.target_node_id, p
  where vs.session_kind='real' and vs.ended_at is not null and vs.anchor_edge_id is not null
    and vs.ended_at >= p.t - interval '210 days' and vs.ended_at <= p.hi
),
r as (select ev.*, unnest(ev.keys) key from ev where w_rule > 0)
select r.ev_id, r.event_type, r.at_, r.bsess, r.duration_ms, r.turns, r.cls, r.key, r.w_rule,
  extract(epoch from (p.t - r.at_))/86400 age,
  r.w_rule*power(2, -(extract(epoch from (p.t - r.at_))/86400)/r.h) w_eff,
  case when r.key in (select key from pin) then 'hit'
       when exists (select 1 from emb where emb.key=r.key and emb.archived_at is not null) then 'archived'
       else 'absent' end status,
  r.qa, r.target_id
from r, p;
-- → 271 rows. Per-node weights: sum w_eff by (key, cls) over status = 'hit'.

-- pair asymmetry rebuild (order-independent per (session_id, target_id))
with e as (select event_type, session_id, coalesce(target_id,'') t from weave_events
  where event_type in ('connection_label_clicked','connection_description_closed','lightbox_opened','lightbox_closed')
    and timestamp >= '2026-07-26T18:15:51.251Z' and timestamp <= '2026-10-04 18:15:55.824614+00'),
g as (select case when event_type like 'connection%' then 'connection' else 'lightbox' end pair, session_id, t,
  count(*) filter (where event_type in ('connection_label_clicked','lightbox_opened')) o,
  count(*) filter (where event_type in ('connection_description_closed','lightbox_closed')) c from e group by 1,2,3)
select pair, sum(o), sum(c), sum(least(o,c)), sum(greatest(o-c,0)), sum(greatest(c-o,0)) from g group by 1;
-- → connection 96|87|87|9|0 ; lightbox 40|38|38|2|0
```
